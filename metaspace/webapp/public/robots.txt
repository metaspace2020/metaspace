# Policy: AI search and assistants may read and cite the site; AI model
# training may not. Search engines are unaffected.
#
# Crawlers pick the most specific User-agent group and ignore the rest, so the
# search/assistant crawlers (Googlebot, Bingbot, OAI-SearchBot, ChatGPT-User,
# Claude-SearchBot, Claude-User, PerplexityBot, Perplexity-User, Applebot,
# DuckAssistBot, meta-externalfetcher) deliberately have no group of their own:
# they fall through to "*" below and are allowed.
#
# robots.txt is voluntary. A crawler that ignores it has to be blocked by
# User-agent in nginx, reactively, from the access log.

Disallow: /db/
Disallow: /mol-images/
